cmake_minimum_required(VERSION 3.20)
project(TinyPTZ VERSION 0.3.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ptz_core STATIC src/camera.cpp src/presets.cpp src/protocol.cpp src/settings.cpp)
target_include_directories(ptz_core PUBLIC src)
target_compile_definitions(ptz_core PUBLIC UNICODE _UNICODE NOMINMAX)
target_link_libraries(ptz_core PUBLIC ole32 oleaut32 strmiids)
add_executable(TinyPTZ WIN32 src/app.cpp src/settings_ui.cpp)
target_link_libraries(TinyPTZ PRIVATE ptz_core comctl32 comdlg32 shell32)
add_executable(ptzctl src/cli.cpp)
target_link_libraries(ptzctl PRIVATE ptz_core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ptz_core)
target_include_directories(core_tests PRIVATE src)
target_compile_definitions(core_tests PRIVATE UNICODE _UNICODE NOMINMAX)
add_executable(settings_ui_tests tests/settings_ui_tests.cpp)
target_link_libraries(settings_ui_tests PRIVATE ptz_core comctl32 gdi32)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
add_test(NAME settings_ui_tests COMMAND settings_ui_tests)
option(BUILD_OBS_PLUGIN "Build the OBS Studio Qt dock plugin (requires matching OBS/Qt SDKs)" OFF)
if(BUILD_OBS_PLUGIN)
 find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
 if(TINY_OBS_SOURCE_DIR AND TINY_OBS_LIBRARY_DIR)
  # build-obs.ps1 prepares official source headers + installed-runtime import libs.
  add_library(OBS::libobs SHARED IMPORTED)
  set_target_properties(OBS::libobs PROPERTIES
   IMPORTED_IMPLIB "${TINY_OBS_LIBRARY_DIR}/obs.lib"
   INTERFACE_INCLUDE_DIRECTORIES "${TINY_OBS_SOURCE_DIR}/libobs")
  add_library(OBS::obs-frontend-api SHARED IMPORTED)
  set_target_properties(OBS::obs-frontend-api PROPERTIES
   IMPORTED_IMPLIB "${TINY_OBS_LIBRARY_DIR}/obs-frontend-api.lib"
   INTERFACE_INCLUDE_DIRECTORIES "${TINY_OBS_SOURCE_DIR}/UI/obs-frontend-api")
 else()
  find_package(libobs REQUIRED CONFIG)
  find_package(obs-frontend-api REQUIRED CONFIG)
 endif()
 # Qt and OBS use the shared CRT; keep the standalone app's static CRT separate.
 add_library(ptz_obs_core STATIC src/camera.cpp src/presets.cpp src/protocol.cpp src/settings.cpp)
 target_include_directories(ptz_obs_core PUBLIC src)
 target_compile_definitions(ptz_obs_core PUBLIC UNICODE _UNICODE NOMINMAX)
 target_link_libraries(ptz_obs_core PUBLIC ole32 oleaut32 strmiids user32)
 add_library(ptz_obs_dock STATIC obs/dock.cpp obs/camera_worker.cpp)
 target_include_directories(ptz_obs_dock PUBLIC obs)
 target_link_libraries(ptz_obs_dock PUBLIC ptz_obs_core Qt6::Widgets)
 target_compile_definitions(ptz_obs_dock PUBLIC QT_NO_KEYWORDS)
 add_library(tiny-ptz MODULE obs/plugin.cpp)
 target_link_libraries(tiny-ptz PRIVATE ptz_obs_dock OBS::libobs OBS::obs-frontend-api)
 add_executable(obs_tests tests/obs_tests.cpp)
 target_link_libraries(obs_tests PRIVATE ptz_obs_dock)
 foreach(target ptz_obs_core ptz_obs_dock tiny-ptz obs_tests)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(${target} PRIVATE /utf-8 /Zc:__cplusplus)
 endforeach()
 add_test(NAME obs_dock_tests COMMAND obs_tests)
 set_tests_properties(obs_dock_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
 install(TARGETS tiny-ptz LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
endif()
