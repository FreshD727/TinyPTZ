cmake_minimum_required(VERSION 3.20)
project(TinyPTZ VERSION 1.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(WIN32)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 set(PTZ_PLATFORM_SOURCE src/camera.cpp)
else()
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Camera backends are available for Windows and Linux.")
 endif()
 set(PTZ_PLATFORM_SOURCE src/camera_linux.cpp src/camera_linux_motion.cpp)
endif()
find_package(Threads REQUIRED)
set(PTZ_CORE_SOURCES ${PTZ_PLATFORM_SOURCE} src/presets.cpp src/protocol.cpp src/settings.cpp)
function(configure_ptz_core target)
 target_include_directories(${target} PUBLIC src)
 target_link_libraries(${target} PUBLIC Threads::Threads)
 if(WIN32)
  target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX)
  target_link_libraries(${target} PUBLIC ole32 oleaut32 strmiids user32)
 endif()
endfunction()
function(qt_target target)
 if(MSVC)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(${target} PRIVATE /utf-8 /Zc:__cplusplus /permissive-)
 endif()
endfunction()
add_library(ptz_core STATIC ${PTZ_CORE_SOURCES})
configure_ptz_core(ptz_core)
option(BUILD_QT_DESKTOP "Build the portable Qt desktop application" ON)
option(BUILD_OBS_PLUGIN "Build the OBS Studio Qt dock plugin" OFF)
include(CTest)
if(BUILD_QT_DESKTOP OR BUILD_OBS_PLUGIN)
 find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets)
 add_library(ptz_obs_core STATIC ${PTZ_CORE_SOURCES})
 configure_ptz_core(ptz_obs_core)
 qt_target(ptz_obs_core)
 add_library(ptz_obs_dock STATIC obs/dock.cpp obs/camera_worker.cpp)
 target_include_directories(ptz_obs_dock PUBLIC obs)
 target_compile_definitions(ptz_obs_dock PUBLIC QT_NO_KEYWORDS)
 target_link_libraries(ptz_obs_dock PUBLIC ptz_obs_core Qt6::Widgets)
 qt_target(ptz_obs_dock)
endif()
if(BUILD_QT_DESKTOP)
  add_library(ptz_window STATIC src/qt_window.cpp)
  target_link_libraries(ptz_window PUBLIC ptz_obs_dock)
  if(WIN32)
   target_link_libraries(ptz_window PUBLIC dwmapi)
  else()
   find_package(Qt6 6.2 REQUIRED COMPONENTS Multimedia)
   find_package(PkgConfig REQUIRED)
   pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
   target_sources(ptz_window PRIVATE src/linux_preview.cpp)
   target_link_libraries(ptz_window PUBLIC Qt6::Multimedia PkgConfig::GSTREAMER)
  endif()
  qt_target(ptz_window)
  add_executable(TinyPTZ WIN32 src/qt_app.cpp)
  set_property(TARGET TinyPTZ PROPERTY AUTORCC ON)
  target_sources(TinyPTZ PRIVATE assets/app_icon.qrc)
  if(WIN32)
   enable_language(RC)
   target_sources(TinyPTZ PRIVATE assets/windows.rc)
   target_include_directories(TinyPTZ PRIVATE assets)
  endif()
  target_link_libraries(TinyPTZ PRIVATE ptz_window)
  qt_target(TinyPTZ)
  if(NOT WIN32)
   install(TARGETS TinyPTZ RUNTIME DESTINATION bin COMPONENT desktop)
   install(FILES packaging/linux/tinyptz.desktop DESTINATION share/applications COMPONENT desktop)
   install(FILES assets/tinyptz.svg DESTINATION share/icons/hicolor/scalable/apps COMPONENT desktop)
   install(FILES assets/tinyptz.png DESTINATION share/icons/hicolor/256x256/apps COMPONENT desktop)
   install(FILES docs/DESKTOP-LINUX.md DESTINATION share/doc/tinyptz RENAME README.md COMPONENT desktop)
   install(FILES LICENSE.txt DESTINATION share/doc/tinyptz COMPONENT desktop)
  endif()
  if(BUILD_TESTING)
   add_executable(desktop_tests tests/desktop_tests.cpp)
   target_link_libraries(desktop_tests PRIVATE ptz_window)
   qt_target(desktop_tests)
   add_test(NAME desktop_tests COMMAND desktop_tests)
   if(WIN32)
    set_tests_properties(desktop_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
   else()
    set_tests_properties(desktop_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
   endif()
  endif()
endif()
if(WIN32)
 if(NOT BUILD_QT_DESKTOP)
  add_executable(TinyPTZ WIN32 src/app.cpp src/settings_ui.cpp)
  target_link_libraries(TinyPTZ PRIVATE ptz_core comctl32 comdlg32 shell32)
 endif()
 add_executable(ptzctl src/cli.cpp)
 target_link_libraries(ptzctl PRIVATE ptz_core)
 if(BUILD_TESTING)
  add_executable(settings_ui_tests tests/settings_ui_tests.cpp)
  target_link_libraries(settings_ui_tests PRIVATE ptz_core comctl32 gdi32)
  add_test(NAME settings_ui_tests COMMAND settings_ui_tests)
 endif()
endif()
if(BUILD_TESTING)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE ptz_core)
 add_test(NAME core_tests COMMAND core_tests)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(linux_backend_tests tests/linux_backend_tests.cpp)
  target_link_libraries(linux_backend_tests PRIVATE ptz_core)
  add_test(NAME linux_backend_tests COMMAND linux_backend_tests)
  # Exercise the real motion implementation with independent native-position
  # and stale-UVC state, without needing a physical camera or test-only APIs.
  add_executable(linux_motion_tests tests/linux_motion_tests.cpp
    src/camera_linux_motion.cpp src/protocol.cpp src/presets.cpp)
  target_include_directories(linux_motion_tests PRIVATE src)
  target_link_libraries(linux_motion_tests PRIVATE Threads::Threads)
  add_test(NAME linux_motion_tests COMMAND linux_motion_tests)
 endif()
endif()
if(BUILD_OBS_PLUGIN)
 if(WIN32 AND TINY_OBS_SOURCE_DIR AND TINY_OBS_LIBRARY_DIR)
  add_library(OBS::libobs SHARED IMPORTED)
  set_target_properties(OBS::libobs PROPERTIES IMPORTED_IMPLIB "${TINY_OBS_LIBRARY_DIR}/obs.lib" INTERFACE_INCLUDE_DIRECTORIES "${TINY_OBS_SOURCE_DIR}/libobs")
  add_library(OBS::obs-frontend-api SHARED IMPORTED)
  set_target_properties(OBS::obs-frontend-api PROPERTIES IMPORTED_IMPLIB "${TINY_OBS_LIBRARY_DIR}/obs-frontend-api.lib" INTERFACE_INCLUDE_DIRECTORIES "${TINY_OBS_SOURCE_DIR}/UI/obs-frontend-api")
 else()
  find_package(libobs REQUIRED CONFIG)
  find_package(obs-frontend-api QUIET CONFIG)
  if(NOT TARGET OBS::obs-frontend-api)
   find_path(OBS_FRONTEND_INCLUDE_DIR obs-frontend-api.h PATH_SUFFIXES obs REQUIRED)
   find_library(OBS_FRONTEND_LIBRARY NAMES obs-frontend-api REQUIRED)
   add_library(OBS::obs-frontend-api UNKNOWN IMPORTED)
   set_target_properties(OBS::obs-frontend-api PROPERTIES IMPORTED_LOCATION "${OBS_FRONTEND_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${OBS_FRONTEND_INCLUDE_DIR}")
  endif()
 endif()
 add_library(tiny-ptz MODULE obs/plugin.cpp)
 set_target_properties(tiny-ptz PROPERTIES PREFIX "" OUTPUT_NAME "tiny-ptz-obs")
 target_link_libraries(tiny-ptz PRIVATE ptz_obs_dock OBS::libobs OBS::obs-frontend-api)
 qt_target(tiny-ptz)
 if(BUILD_TESTING)
  add_executable(obs_tests tests/obs_tests.cpp)
  target_link_libraries(obs_tests PRIVATE ptz_obs_dock)
  qt_target(obs_tests)
  add_test(NAME obs_dock_tests COMMAND obs_tests)
  set_tests_properties(obs_dock_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
   add_library(obs_frontend_stub SHARED tests/obs_frontend_stub.cpp)
   set_target_properties(obs_frontend_stub PROPERTIES OUTPUT_NAME obs-frontend-api SOVERSION 0 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/module-host")
   target_link_libraries(obs_frontend_stub PRIVATE OBS::libobs Qt6::Widgets)
   add_executable(obs_module_tests tests/obs_module_tests.cpp)
   target_link_libraries(obs_module_tests PRIVATE obs_frontend_stub ptz_obs_dock OBS::libobs)
   add_test(NAME obs_module_tests COMMAND obs_module_tests $<TARGET_FILE:tiny-ptz>)
   set_tests_properties(obs_module_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
 endif()
 install(TARGETS tiny-ptz LIBRARY DESTINATION bin/64bit RUNTIME DESTINATION bin/64bit)
 install(FILES obs/data/presets.ptz DESTINATION data)
 if(WIN32)
  install(FILES docs/OBS-PLUGIN.md DESTINATION . RENAME README.md)
 else()
  install(FILES docs/OBS-LINUX.md DESTINATION . RENAME README.md)
 endif()
 install(FILES LICENSE.txt DESTINATION . OPTIONAL)
endif()
