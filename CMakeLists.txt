cmake_minimum_required(VERSION 3.20)
project(TinyPTZ VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ptz_core STATIC src/camera.cpp src/presets.cpp src/protocol.cpp src/settings.cpp)
target_include_directories(ptz_core PUBLIC src)
target_compile_definitions(ptz_core PUBLIC UNICODE _UNICODE NOMINMAX)
target_link_libraries(ptz_core PUBLIC ole32 oleaut32 strmiids)
add_executable(TinyPTZ WIN32 src/app.cpp src/settings_ui.cpp)
target_link_libraries(TinyPTZ PRIVATE ptz_core comctl32 comdlg32 shell32)
add_executable(ptzctl src/cli.cpp)
target_link_libraries(ptzctl PRIVATE ptz_core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ptz_core)
target_include_directories(core_tests PRIVATE src)
target_compile_definitions(core_tests PRIVATE UNICODE _UNICODE NOMINMAX)
add_executable(settings_ui_tests tests/settings_ui_tests.cpp)
target_link_libraries(settings_ui_tests PRIVATE ptz_core comctl32 gdi32)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
add_test(NAME settings_ui_tests COMMAND settings_ui_tests)
option(BUILD_OBS_PLUGIN "Build the optional OBS Studio hotkey plugin (requires libobs SDK)" OFF)
if(BUILD_OBS_PLUGIN)
 find_package(libobs REQUIRED)
 add_library(tiny-ptz MODULE obs/plugin.cpp)
 target_link_libraries(tiny-ptz PRIVATE ptz_core OBS::libobs)
endif()
